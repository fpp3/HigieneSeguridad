% --- SLIDE 17: CIANURO DE HIDRÓGENO ---
\begin{frame}{Asfixia a Nivel Celular: Cianuro de Hidrógeno ($HCN$)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \small
      Veneno de acción celular directa y fulminante:
      \begin{itemize}
        \scriptsize
        \item \textbf{Origen cotidiano:} Se libera al arder espumas de poliuretano, aislaciones acústicas y térmicas, alfombras y nylon.
        \item \textbf{Bloqueo interno:} Entra a las células y anula su capacidad de procesar el oxígeno que llega por sangre.
        \item \textbf{Ahogo celular:} Aunque la sangre transporte $O_2$, la célula no puede consumirlo y muere en minutos.
        \item \textbf{Sinergia letal con $CO$:} El $CO$ corta la entrega y el $HCN$ apaga el motor celular $\implies$ \textbf{Desmayo en $<3$ minutos}.
      \end{itemize}
    \end{column}

% --- SLIDE 23: CARGA DE FUEGO FÓRMULA ---
\begin{frame}{Determinación de la Carga de Fuego ($Q_f$)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Fórmula Oficial del Anexo VII}}
        \begin{equation*}
          Q_f = \frac{\sum (M_i \cdot PC_i)}{S \cdot 4400} \quad \left[\frac{\kg}{\m^2}\right]
        \end{equation*}
      \end{block}
      \scriptsize
      $\bullet$ $M_i$: Masa de cada material combustible ($\kg$).\\
      $\bullet$ $PC_i$: Poder calorífico inferior ($\mathrm{kcal/kg}$).\\
      $\bullet$ $S$: Superficie útil de piso del sector ($\m^2$).\\
      $\bullet$ $4400\,\mathrm{kcal/kg}$: Poder calorífico de la madera patrón.
      \vspace{0.1cm}
      \begin{alertblock}{\textbf{Cálculo del Informe}}
        \tiny Depósito de $600\,\m^2$ con embalajes y plásticos:\\
        $\sum (M_i \cdot PC_i) = 33.7 \times 10^6\,\mathrm{kcal} \implies \mathbf{Q_f = 12.77\,\kg/\m^2}$\\
        $\rightarrow$ Rango reglamentario: \textbf{$Q_f \le 15\,\kg/\m^2$} (Carga de fuego baja).
