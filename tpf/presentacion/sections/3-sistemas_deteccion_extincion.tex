% --- SLIDE 30: DETECTOR ÓPTICO ---
\begin{frame}{Detector Óptico Puntual de Humo (Efecto Tyndall)}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \small
      Principio físico de dispersión de la luz en cámara oscura:
      \begin{itemize}
        \scriptsize
        \item Dispone de una cámara laberíntica que bloquea la luz ambiental exterior pero permite la libre circulación del aire.
        \item Aloja un LED emisor infrarrojo y un fotodiodo receptor colocado en ángulo diedro ($90^\circ\text{ a }120^\circ$).
        \item \textbf{En reposo:} El haz se propaga en línea recta sin incidir sobre el fotodiodo receptor (corriente nula).
        \item \textbf{En alarma:} Las partículas de humo dispersan los fotones hacia el fotodiodo por efecto Tyndall.
        \item \textbf{Especialidad:} Incendios lentos o latentes con partículas visibles de $0.3\text{ a }10\,\mu\mathrm{m}$ (PVC, celulosa).
      \end{itemize}
    \end{column}

% --- SLIDE 31: DETECTOR IÓNICO ---
\begin{frame}{Detector Iónico de Humo por Americio-241 ($^{241}\mathrm{Am}$)}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \small
      Principio de conducción eléctrica en gases ionizados:
      \begin{itemize}
        \scriptsize
        \item Contiene una microfuente radiactiva encapsulada de $^{241}\mathrm{Am}$ emisora de partículas alfa ($\alpha$).
        \item Las partículas $\alpha$ ionizan las moléculas de aire ($N_2, O_2$), separándolas en iones positivos y electrones libres.
        \item Se establece una \textbf{corriente basal de reposo} entre dos placas cargadas a una diferencia de potencial fija.
        \item Al ingresar partículas submicrónicas de humo ($<0.3\,\mu\mathrm{m}$), los iones se adhieren a ellas, disminuyendo drásticamente la corriente y disparando la alarma.
        \item \textbf{Especialidad:} Fuegos vivos de rápida combustión con llama abierta y aerosoles invisibles.
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \resizebox{0.85\columnwidth}{!}{%

% --- SLIDE 34: TÉRMICA, CABLE SENSOR Y LLAMA ---
\begin{frame}{Detección Térmica, Cable Sensor Lineal y Detectores de Llama}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Detectores Térmicos y Cable Lineal}}
        \scriptsize
        $\bullet$ \textbf{Termostáticos y Termovelocimétricos:} Disparo por umbral fijo ($57\,\Celsius$) o tasa de elevación rápida ($>8\,\Celsius/\mathrm{min}$).\\
        $\bullet$ \textbf{Cable Sensor Lineal Térmico:} Dos conductores bajo tensión mecánica separados por polímero termosensible; el aislante se funde ante sobrecalentamiento produciendo cortocircuito supervisado.\\
        $\bullet$ \textbf{Aplicación:} Bandejas portacables, túneles y subestaciones.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Detectores Ópticos de Llama}}
        \scriptsize
        $\bullet$ Sensores optoelectrónicos sintonizados al espectro infrarrojo ($\mathrm{IR}$ a $4.3\,\mu\mathrm{m}$, emisión de $CO_2$) y ultravioleta ($\mathrm{UV}$ a $0.2\,\mu\mathrm{m}$).\\
        $\bullet$ Algoritmos de parpadeo (\textit{flicker}) para descartar radiación solar.\\
        $\bullet$ Tiempo de respuesta inferior a \textbf{segundos}.\\
        $\bullet$ \textbf{Aplicación:} Depósitos de solventes y turbinas.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --- SLIDE 37: PQS ABC Y BC ---
\begin{frame}{Extintores de Polvo Químico Seco (PQS ABC y BC)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{PQS Triclase ABC (Fosfato Monoamónico)}}
        \scriptsize
        $\bullet$ Base de $\mathrm{NH_4H_2PO_4}$ finamente micronizado con agentes hidrofugantes siliconados.\\
        $\bullet$ \textbf{En fuegos Clase A:} Descompone a $>180\,\Celsius$ formando una capa vidriada de ácido metafosfórico que obtura los poros y aísla el oxígeno.\\
        $\bullet$ \textbf{En fuegos Clase B:} Captura radicales libres e interrumpe la reacción en cadena.\\
        $\bullet$ \textbf{Efecto adverso severo:} Altamente corrosivo e higroscópico; arruina circuitos electrónicos delicados.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{PQS Clase BC (Bicarbonato de Sodio/Potasio)}}
        \scriptsize
        $\bullet$ Formulado a base de $\mathrm{NaHCO_3}$ o bicarbonato de potasio $\mathrm{KHCO_3}$.\\
        $\bullet$ No forma costra vidriada adherente; limpieza más sencilla en motores y maquinaria mecánica.\\
        $\bullet$ Excelente eficacia en derrames de líquidos hidrocarburos y gases a presión.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Dióxido de Carbono ($CO_2$) y Agentes Limpios en Electrónica}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \small
      Los agentes limpios preservan los circuitos sin dejar residuos:
      \begin{itemize}
        \scriptsize
        \item \textbf{Dióxido de Carbono ($CO_2$):}
        \begin{itemize}
          \tiny
          \item Almacenado a $50\text{--}60\,\mathrm{bar}$ en cilindros sin costura; se descarga en nieve carbónica a $\mathbf{-78.5\,\Celsius}$.
          \item Gas inerte, no conductor de electricidad y de evaporación limpia sin humedad.
          \item Tobera difusora cónica no metálica para evitar descargas estáticas.
        \end{itemize}
        \item \textbf{Fluidos Fluorados (FK-5-1-12 / Novec 1230):}
        \begin{itemize}
          \tiny
          \item Líquido vaporizante con cero potencial de agotamiento de ozono.
          \item Absorbe calor e interrumpe la combustión sin provocar choque térmico extremo en microprocesadores.
        \end{itemize}

% --- SLIDE 39: POTENCIAL EXTINTOR ENSAYOS ---
\begin{frame}{Potencial Extintor y Ensayos de Homologación (IRAM 3517-2)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Rating Clase A ($1\mathrm{A}$ a $40\mathrm{A}$)}}
        \scriptsize
        $\bullet$ Ensayado sobre una pira normalizada de listones de madera de pino secados en estufa y dispuestos en capas cruzadas.\\
        $\bullet$ \textbf{Definición de 1A:} Capacidad de extinguir una pira de madera equivalente a $1.25\,\mathrm{gal}$ ($5\,\mathrm{L}$) de agua aplicada por un operador calificado.\\
        $\bullet$ Un matafuego con rating $4\mathrm{A}$ apaga cuatro veces más madera que uno de $1\mathrm{A}$.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Rating Clase B ($1\mathrm{B}$ a $120\mathrm{B}$)}}
        \scriptsize
        $\bullet$ Ensayado sobre bateas de acero cuadradas con nafta heptano y combustión libre.\\
        $\bullet$ \textbf{Definición:} El número indica el área en pies cuadrados ($1\,\mathrm{ft^2} \approx 0.09\,\m^2$) de líquido inflamable que un operador no experto es capaz de sofocar con seguridad.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --- SLIDE 41: SPRINKLERS E INUNDACIÓN GASEOSA ---
\begin{frame}{Rociadores Automáticos (\textit{Sprinklers}) e Inundación Gaseosa}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Rociadores Automáticos (NFPA 13)}}
        \scriptsize
        $\bullet$ Red de cañerías presurizadas permanentemente bajo techo.\\
        $\bullet$ \textbf{Ampolla termosensible de cuarzo:} Contiene un líquido con burbuja de vapor que expande por calor y estalla al alcanzar la temperatura nominal ($68\,\Celsius$ rojo, $79\,\Celsius$ amarillo, $93\,\Celsius$ verde).\\
        $\bullet$ \textbf{Acción puntual:} Se abre únicamente el rociador ubicado sobre el foco, evitando anegamientos innecesarios.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Inundación Total por Gas Limpio}}
        \scriptsize
        $\bullet$ Batería de cilindros de alta presión vinculada a toberas de descarga volumétrica en recinto cerrado.\\
        $\bullet$ Activación cruzada de dos zonas de detectores para evitar disparos accidentales.\\
        $\bullet$ Descarga integral en menos de $\mathbf{10\,\mathrm{s}}$, extinguiendo el fuego por inhibición química o desplazamiento controlado sin daño en servidores ni peligro letal para operarios.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
